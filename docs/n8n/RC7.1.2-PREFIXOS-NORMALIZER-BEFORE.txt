/** RC7.1.2. Pure contract adapter; legacy contains the published message formatting. */
function normalizeRc712(body, legacy) {
  const b = body && typeof body === 'object' && !Array.isArray(body) ? body : {};
  const identifier = (v, max = 160) =>
    typeof v === 'string' && v.length > 0 && v.length <= max && /^[A-Za-z0-9_.:-]+$/.test(v);
  const text = (v) => typeof v === 'string' && v.trim().length > 0;
  const esc = (v) =>
    String(v ?? 'indisponível')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;');
  const base = {
    EVENTO: b.event ?? null,
    EVENT_ID: b.eventId ?? null,
    PROFILE_ID: b.profileId ?? null,
    EPISODIO_ID: b.episodeId ?? null,
    REAL_ACTIVATED: false,
    DETECTION_BLOCKED: false,
    MITIGATION_STATUS: false,
    RC712_VALID: false,
    RC712_ROUTE: null,
  };
  const reject = (reason) => ({ ...base, MOTIVO: reason });
  if (!identifier(b.profileId) || !identifier(b.eventId, 256)) return reject('IDENTITY_INVALID');
  if (
    b.occurredAt !== null &&
    b.occurredAt !== undefined &&
    (!text(b.occurredAt) || !Number.isFinite(Date.parse(b.occurredAt)))
  )
    return reject('TIMESTAMP_INVALID');
  const node = (v) => Number.isInteger(v) && v > 0 && v <= 65535;
  const scope = () =>
    text(b.policy) &&
    b.policy.length <= 128 &&
    /^[A-Za-z0-9_.:-]+$/.test(b.policy) &&
    node(b.bogonNode) &&
    node(b.mitigationNode ?? b.node) &&
    b.bogonNode !== (b.mitigationNode ?? b.node) &&
    (b.node == null || b.node === (b.mitigationNode ?? b.node)) &&
    typeof b.rt === 'string' &&
    /^\d{1,10}:\d{1,10}$/.test(b.rt) &&
    ['IPV4', 'IPV6'].includes(b.addressFamily);
  const readback = (state) => {
    const raw = b.readback ?? b.readBack;
    const runtime = b.mitigationState ?? b.state;
    if (runtime != null && runtime !== state) return false;
    if (raw === undefined || raw === null) return true; // Candidate sends verified + identity, not a readback object.
    if (
      typeof raw !== 'object' ||
      Array.isArray(raw) ||
      raw.verified !== true ||
      raw.state !== state ||
      raw.policy !== b.policy
    )
      return false;
    if (raw.bogonNode !== undefined && raw.bogonNode !== b.bogonNode) return false;
    if (raw.mitigationNode !== undefined && raw.mitigationNode !== (b.mitigationNode ?? b.node))
      return false;
    return raw.rt === undefined || raw.rt === b.rt;
  };
  if (b.event === 'MITIGATION_VERIFIED') {
    if (
      b.simulation !== false ||
      b.verified !== true ||
      !scope() ||
      !readback('MITIGATED') ||
      ('remove_confirm|' + b.profileId).length > 64
    )
      return reject('ACTIVATION_UNVERIFIED_OR_SCOPE_INVALID');
    return {
      ...legacy(b),
      ...base,
      RC712_VALID: true,
      RC712_ROUTE: 'ACTIVATED',
      REAL_ACTIVATED: true,
    };
  }
  if (b.event === 'MITIGATION_DETECTION_BLOCKED') {
    if (
      b.simulation !== true ||
      b.verified !== false ||
      !identifier(b.episodeId) ||
      !text(b.blockedReason)
    )
      return reject('DETECTION_INVALID');
    if (b.direction != null && !['rx', 'tx'].includes(b.direction))
      return reject('DIRECTION_INVALID');
    for (const field of ['trafficGbps', 'thresholdGbps', 'trafficMbps', 'thresholdMbps']) {
      if (
        b[field] != null &&
        (typeof b[field] !== 'number' || !Number.isFinite(b[field]) || b[field] < 0)
      )
        return reject('RATE_INVALID');
    }
    return {
      ...legacy(b),
      ...base,
      RC712_VALID: true,
      RC712_ROUTE: 'DETECTION',
      DETECTION_BLOCKED: true,
    };
  }
  if (b.event === 'MITIGATION_STATUS_RESEND') {
    if (b.simulation !== false || b.verified !== true || !scope() || !readback('MITIGATED'))
      return reject('RESEND_INVALID');
    const old = legacy(b);
    if (old.RESEND_VALIDO !== true) return reject('RESEND_READBACK_REQUIRED');
    return {
      ...old,
      ...base,
      RC712_VALID: true,
      RC712_ROUTE: 'STATUS',
      MITIGATION_STATUS: true,
      REENVIO: true,
      statusText: old.telegramText,
    };
  }
  const removed = b.event === 'MITIGATION_REMOVED_VERIFIED';
  const failed = b.event === 'MITIGATION_FAILED';
  if (!removed && !failed) return reject('UNKNOWN_EVENT');
  if (b.simulation !== false || b.verified !== removed)
    return reject('STATUS_VERIFICATION_INVALID');
  if (removed && (!scope() || !readback('NORMAL')))
    return reject('REMOVAL_UNVERIFIED_OR_SCOPE_INVALID');
  const lines = [
    removed
      ? '<b>RETIRADA MANUAL VERIFICADA</b>'
      : '<b>FALHA / ESTADO INCERTO — REVISÃO NECESSÁRIA</b>',
    '',
  ];
  for (const [label, value] of [
    ['Cliente', b.customer],
    ['Perfil', b.profileId],
    ['Equipamento', b.device],
    ['Interface', b.interfaceName ?? b.interface],
    ['Policy', b.policy],
    ['Família', b.addressFamily],
    ['RT', b.rt],
    ['Node BOGONS', b.bogonNode],
    ['Node mitigação', b.mitigationNode ?? b.node],
    ['Operador', b.operatorRef ?? b.confirmedBy ?? b.operator],
    ['Horário', b.occurredAt],
  ])
    lines.push('<b>' + label + ':</b> ' + esc(value));
  lines.push(
    '',
    removed
      ? 'Readback de retirada confirmado pela API. Retirada exclusivamente manual.'
      : 'Estado físico não confirmado. Não inferir sucesso nem executar retirada automática; revisar readback e journal.',
  );
  return {
    ...base,
    RC712_VALID: true,
    RC712_ROUTE: 'STATUS',
    MITIGATION_STATUS: true,
    statusText: lines.join('\n'),
  };
}

    const __rBogon = __rBody.bogonNode ?? null;
    const __rMit = __rBody.mitigationNode ?? __rBody.node ?? null;

      __rMotivo = 'PROFILE_INVALIDO';
    } else if (__rSimulado) {
      __rMotivo = 'SIMULACAO';
    } else if (!__rVerificado) {
      __rMotivo = 'NAO_VERIFICADO';
    } else if (!__rEstadoOk) {
      __rMotivo = 'ESTADO_NAO_MITIGADO';
    } else if (__rRb === null || typeof __rRb !== 'object') {

    } else if (__rRb.verified !== true) {

    } else if (__rUp(__rRb.state) !== 'MITIGATED') {

    } else if (__rRb.policy !== (__rBody.policy ?? null)) {

      );

    __rLines.push('');

    __rLines.push(
      'ℹ️ Estado <b>já ativo</b> no equipamento — reenvio operacional; <b>não</b> é uma nova ativação.'

  const realActivated =

    simulation === false &&
    verified === true &&

    CLIENTE: b.customer ?? null,

    ORIGEM_DIRECAO: b.directionSource ?? null,

  };
